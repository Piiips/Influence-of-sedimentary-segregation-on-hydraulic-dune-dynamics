\section{Introducción}

\begin{frame}{Contenidos}
Puntos centrales del desarrollo matemático:
\begin{itemize}
  \item Marco comóvil y geometría de la duna
  \item Cambio de variable $Q = h\,\phi_s$ y forma conservativa
  \item Campo de velocidades libre de divergencia
  \item Flujos de Trewhela con capa activa confinada
  \item Condiciones de borde: deposición por avalanchas
  \item Método numérico: WENO5 + MUSCL/Rusanov
  \item Extensión a geometría curva
\end{itemize}
\end{frame}

\begin{frame}{Introducción y alcance}
Se simula la evolución de la concentración volumétrica de finos $\phi_s(x,\eta,t)\in[0,1]$ en una duna bidispersa (finos $d_s$, gruesos $d_l>d_s$), a partir del esquema de advección--segregación--difusión de Trewhela, Ancey \& Gray (2021) y la fenomenología de \emph{grain-flows} de Kleinhans (2004).

\vspace{6pt}
\textbf{Cinco adaptaciones centrales:}
\begin{enumerate}
  \item Cambio de variable $Q=h\phi_s$ y coordenadas $\sigma$
  \item Confinamiento de la capa activa y desacople del corte
  \item Piso de presión que regulariza la segregación
  \item Deposición por avalanchas discretas en el lee
  \item Esquema de alto orden: WENO5 + MUSCL/Rusanov
\end{enumerate}

\vspace{4pt}
$R = d_l/d_s = 3.33$ ($d_l=1.0$~mm, $d_s=0.3$~mm), $\phi_s=0.70$ constante.
\end{frame}

\begin{frame}{Marco comóvil y geometría de la duna}
La duna migra a celeridad $c_{\mathrm{mig}}$; se trabaja en el \textbf{marco comóvil} ($x_{\mathrm{lab}}=x+c_{\mathrm{mig}}t$) donde su forma es estacionaria.

\vspace{4pt}
Topografía triangular a trozos:
\begin{equation*}
    h(x) = \begin{cases}
        H_{\mathrm{base}} + H_d\dfrac{x}{x_{\mathrm{crest}}}, & 0\le x \le x_{\mathrm{crest}} \ \text{(stoss)}\\[8pt]
        H_{\mathrm{base}} + H_d\Big(1-\dfrac{x-x_{\mathrm{crest}}}{L_{\mathrm{lee}}}\Big), & x_{\mathrm{crest}}<x\le L_{\mathrm{dune}} \ \text{(lee)}
    \end{cases}
\end{equation*}
con $H_d=8$~mm, $L_{\mathrm{dune}}=100$~mm, ángulo de reposo del lee $\alpha_{\mathrm{lee}}=30^\circ$.

\vspace{6pt}
\textbf{Escala de demostración:} $c_{\mathrm{mig}}=10\times c_{\mathrm{mig,fis}}$ acelera el reciclaje sin alterar la física del \emph{sorting}.
\end{frame}

\section{Cambio de var.}

\begin{frame}{Coordenadas $\sigma$ y masa conservada}
Coordenada vertical normalizada $\eta = z/h(x)\in[0,1]$ ($\eta=0$ base, $\eta=1$ superficie). La masa total de finos:
\begin{equation*}
    M_{\mathrm{finos}} = \int_0^L\!\!\int_0^{h(x)}\!\phi_s\,\dd z\,\dd x
    = \int_0^L\!\!\int_0^1 \underbrace{h(x)\,\phi_s}_{\displaystyle Q}\,\dd\eta\,\dd x .
\end{equation*}

\vspace{6pt}
Resolver directamente para $Q=h\,\phi_s$ (espesor equivalente de finos) hace que la masa de cada celda, $Q\,\dd x\,\dd\eta$, sea invariante ante cualquier deformación de $h$: \textbf{conservación estricta de masa}.
\end{frame}

\begin{frame}{Forma conservativa en $\sigma$}
Partiendo de la conservación cartesiana de la fase fina y aplicando el cambio de variable $(x,z,t)\to(x,\eta,t)$ con $Q=h\phi_s$:
\begin{equation*}
    \boxed{\;\pd{Q}{t} + \pd{(u\,Q)}{x}
    + \pd{}{\eta}\Big[\, F_{\mathrm{adv},\eta} + F_{\mathrm{seg},\eta} - F_{\mathrm{diff},\eta} \,\Big] = 0\;}
\end{equation*}
con los tres flujos verticales
\begin{align*}
    F_{\mathrm{adv},\eta} &= \big(w-\eta\,\partial_t h-\eta\,u\,\partial_x h\big)\phi_s ,\\
    F_{\mathrm{seg},\eta} &= -\,f_{sl}\,\phi_s(1-\phi_s),\\
    F_{\mathrm{diff},\eta} &= \frac{D_{sl}}{h}\,\pd{\phi_s}{\eta}.
\end{align*}
La derivación completa (regla de la cadena + regla del producto para introducir $Q$) reagrupa todos los términos topográficos bajo un único operador $\partial/\partial\eta$.
\end{frame}

% ------------------------------------------------------------------
\section{Velocidades}

\begin{frame}{Campo de velocidades libre de divergencia}
Se derivan $u,w$ de una función de corriente $\psi=U_0H_{\mathrm{base}}\eta^{m+1}g(x,t)$, con $\eta^m$ ($m=3$) concentrando el transporte cerca de la superficie ($\eta=1$):
\begin{align*}
    u(x,\eta,t) &= \frac{U_0 H_{\mathrm{base}}}{h(x)}(m+1)\eta^m g(x,t) \;-\; c_{\mathrm{mig}},\\
    \omega_{\mathrm{code}}(x,\eta,t) &= c_{\mathrm{mig}}\,\eta\,\frac{\dd h}{\dd x} - U_0 H_{\mathrm{base}}\,\eta^{m+1}\,\pd{g}{x}.
\end{align*}
El término $-c_{\mathrm{mig}}$ es la recirculación del marco comóvil: produce \textbf{erosión en el stoss} y \textbf{deposición en el lee}. Esta construcción satisface analíticamente $\partial_x(hu)+\partial_\eta\omega_{\mathrm{code}}=0$.
\end{frame}

\begin{frame}{Tasa de corte desacoplada y confinada}
En el modelo clásico, bajar $U_0$ (advección) también apaga la segregación. Aquí se introduce una escala de corte independiente $U_{\mathrm{shear}}\gg U_0$:
\begin{equation*}
    \dot\gamma(x,\eta,t) =
    \underbrace{\frac{U_{\mathrm{shear}}H_{\mathrm{base}}}{h_{\mathrm{eff}}^2}m(m+1)\eta^{m-1}g(x,t)}_{\text{perfil de corte}}
    \cdot \underbrace{W_{\mathrm{act}}(x,\eta)}_{\text{confinamiento}}
\end{equation*}
con $h_{\mathrm{eff}}=\max(h,2.5\delta_a)$ evitando divergencia en pies delgados.

\vspace{4pt}
\textbf{Ventana de capa activa} (escalón $\tanh$, no gaussiano: evita lavar láminas enterradas tras miles de sub-pasos):
\begin{equation*}
    W_{\mathrm{act}} = \tfrac12\Big[1-\tanh\Big(\tfrac{z_\downarrow-\delta_a}{w_{\tanh}}\Big)\Big], \quad \delta_a=1.5\text{ mm}.
\end{equation*}
\end{frame}

\begin{frame}{Segregación y difusión}
\begin{align*}
    \bar d(\phi_s) &= (1-\phi_s)d_l+\phi_s d_s, \\[2pt]
    F(R,\phi_s) &= (R-1)+E(1-\phi_s)(R-1)^2, \\[2pt]
    f_{sl} &= \frac{B\,\dot\gamma\,\bar d^2}{C\bar d+p(x,\eta)}\,F(R,\phi_s), \qquad D_{sl} = A\,\dot\gamma\,\bar d^2 .
\end{align*}
El flujo segregativo es $F_{\mathrm{seg}}=-f_{sl}\,\phi_s(1-\phi_s)$; el factor $\phi_s(1-\phi_s)$ apaga la segregación en zonas puras.

\vspace{4pt}
{\small\textbf{Constantes de Trewhela et al.\ (2021), Tabla 3:} $A=0.108$, $B=0.3744$, $C=0.2712$, $E=2.0957$.\par}
\end{frame}

\begin{frame}{Piso de presión (regularización)}
Regulariza $f_{sl}$ cerca de la superficie, donde la presión litostática $\to 0$:
\begin{equation*}
    p(x,\eta) = \nu h(x)(1-\eta) + p_{\mathrm{floor}}, \qquad p_{\mathrm{floor}}=3\nu\delta_a .
\end{equation*}
con $\nu=0.6$ y $U_{\mathrm{shear}}=0.015$~m/s, parámetros propios del proyecto (no tomados del paper de Trewhela et al.).

\vspace{6pt}
En la corrida documentada (log), esto da $f_{sl,\max}\approx3.3$~mm/s, con $n_{\mathrm{sub}}=318$ (versión triangular) y $n_{\mathrm{sub}}=142$ (versión curva).
\end{frame}

\section{Bordes y numérico}

\begin{frame}{Deposición por avalanchas discretas}
\textbf{Base} ($\eta=0$): impermeable. \textbf{Superficie} ($\eta=1$): en el stoss (erosión) sale con la concentración local; en el lee (deposición) entra con
\begin{equation*}
    \phi_{\mathrm{in}}(x,t) = \mathrm{clip}\big(\lambda(t)\,P_{\mathrm{spat}}(x)\,P_{\mathrm{temp}}(t),\,0,\,1\big),
\end{equation*}
donde $\lambda(t)$ cierra el balance masa erosionada = masa depositada, y $P_{\mathrm{spat}}$ impone grueso al pie / fino a la cresta (Kleinhans, 2004).

\vspace{4pt}
\textbf{Novedad — pulso agudizado} (avalanchas discretas, no seno suave):
\begin{equation*}
    P_{\mathrm{temp}}(t) = 1 + A_P\,\mathrm{sign}(\sin\theta)\,|\sin\theta|^{p_{\mathrm{sharp}}}, \quad \theta=\tfrac{2\pi t}{T_{\mathrm{period}}},
\end{equation*}
con $A_P=0.90$, $p_{\mathrm{sharp}}=0.5$: mayor $A_P$ y menor $p_{\mathrm{sharp}}$ agudizan la alternancia de láminas.

\vspace{2pt}
Dominio periódico en $x$: recicla grueso del pie del lee al stoss.
\end{frame}

\begin{frame}{Condición inicial}
La duna arranca \textbf{homogénea}:
\begin{equation*}
    \phi_s(x,\eta,0) = \phi_{s,\mathrm{target}} = 0.70 \quad \text{en todo el dominio},
\end{equation*}
sin gradación ni estructura sembrada.

\vspace{6pt}
Todo el patrón interno --armadura de gruesos superficial, \emph{lag} basal de gruesos, láminas-- \textbf{emerge} de la dinámica de segregación confinada más la deposición por avalanchas.
\end{frame}

\begin{frame}{Método numérico: splitting + alto orden}
\emph{Operator splitting}: paso horizontal grande $\Delta t_h$ + sub-ciclo vertical de $n_{\mathrm{sub}}$ pasos.

\vspace{4pt}
\textbf{Horizontal (SSP-RK2 + WENO5):} transporte $-\partial_x(uQ)$ con reconstrucción de 5.\textsuperscript{o} orden en las caras (pesos no lineales $\omega_k$ vía indicadores de suavidad); reemplazó a MUSCL de 2.\textsuperscript{o} orden porque este difuminaba las láminas advectadas.

\vspace{4pt}
\textbf{Vertical (MUSCL + Rusanov):} pendientes limitadas por \emph{minmod}; advección lineal con upwind exacto; segregación no lineal con flujo de Rusanov; difusión con diferencias centradas.

\vspace{4pt}
\textbf{CFL:}
\begin{equation*}
    \small \Delta t_h = 0.5\frac{\dd x}{u_{\max}}, \qquad
    \Delta t_v = \min\Big(0.5\tfrac{\dd\eta\min h}{w_{\eta,\max}+f_{sl,\max}},\ 0.45\tfrac{(\dd\eta\min h)^2}{D_{sl,\max}}\Big).
\end{equation*}

\small Malla $N_x=160$, $N_z=40$, $t_{\max}=2400$~s; recorte conservativo + corrección multiplicativa global mantienen $\langle\phi_s\rangle=0.700000$ exacto.
\end{frame}

\begin{frame}{Adaptaciones para la versión curva}
La versión curva comparte \emph{toda} la física y numérica anteriores; \textbf{sólo} cambia $h(x)$:
\begin{enumerate}
  \item \textbf{Stoss convexo:} $h_{\mathrm{stoss}}=H_{\mathrm{base}}+H_d(x/x_{\mathrm{crest}})^{p_{\mathrm{stoss}}}$, $p_{\mathrm{stoss}}=0.55$.
  \item \textbf{Suavizado $C^1$ de la cresta:} convolución con filtro Gaussiano periódico, $\tilde h=\mathcal G_{\sigma_s}\ast h_{\mathrm{tent}}$.
  \item \textbf{Reescalado} para conservar $H_d$ en la cresta; $\dd h/\dd x$ por diferencias centradas periódicas.
\end{enumerate}
\vspace{4pt}
\textbf{Motivación:} la topografía triangular tiene $\dd h/\dd x$ discontinua en la cresta, contaminando $\omega_{\mathrm{code}}$ y el corte de segregación cerca de ella. Con $\dd h/\dd x$ continua, ese artefacto desaparece.
\end{frame}

\end{frame}

\begin{frame}{Síntesis}
\begin{itemize}
  \item El cambio $Q=h\phi_s$ en coordenadas $\sigma$ da conservación estricta de masa ante topografía variable.
  \item Desacoplar y confinar la tasa de corte separa la advección lenta de la segregación rápida de la capa activa.
  \item El piso de presión regulariza $f_{sl}$ sin alterar cualitativamente el \emph{sorting}.
  \item Las avalanchas discretas en el borde generan la laminación observada; el dominio periódico cierra el reciclaje de masa.
  \item WENO5 + MUSCL/Rusanov preservan láminas que esquemas de 2.\textsuperscript{o} orden difuminan.
  \item La geometría curva comparte la misma física y numérica que la triangular; solo cambia $h(x)$ (observación cualitativa, sin métrica comparativa en los logs).
\end{itemize}

\vspace{8pt}
\small \textbf{Referencias:} Trewhela, Ancey \& Gray (2021); Kleinhans (2004).
\end{frame}
